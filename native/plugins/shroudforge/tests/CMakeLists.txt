cmake_minimum_required(VERSION 3.24)
project(kiln_sf_review_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_executable(eml_abi eml_abi_test.cpp)
add_executable(sfhost host.cpp)
target_include_directories(sfhost PRIVATE ../../../core/abi)
foreach(name IN ITEMS runtime runtime_feed runtime_feed_partial good_kfc old_kfc bad_kfc eml bootstrap)
  add_library(${name} SHARED fake.cpp)
  target_include_directories(${name} PRIVATE ../../../core/abi)
  target_compile_definitions(${name} PRIVATE "TEST_${name}=1")
endforeach()
